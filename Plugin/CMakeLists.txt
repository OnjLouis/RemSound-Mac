cmake_minimum_required(VERSION 3.22)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Minimum macOS")
project(RemSoundPlugin VERSION 6.2.4 LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
add_library(RemSoundPluginHostRuntime SHARED Source/PluginHostRuntime.cpp)
target_include_directories(RemSoundPluginHostRuntime PRIVATE Source)
target_link_libraries(RemSoundPluginHostRuntime PRIVATE Threads::Threads)
set_target_properties(RemSoundPluginHostRuntime PROPERTIES
    OUTPUT_NAME RemSoundPluginHost-v1 INSTALL_NAME_DIR "@rpath"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/host-runtime/$<CONFIG>"
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG 91ad83ae34a81e0833b1a2b0866f54846370ae53)
FetchContent_Declare(clap_juce GIT_REPOSITORY https://github.com/free-audio/clap-juce-extensions.git
    GIT_TAG 55525c9858d4b25687be7759a5e0f70eccef218e)
FetchContent_MakeAvailable(JUCE clap_juce)
juce_add_plugin(RemSoundPlugin
    COMPANY_NAME "Ed Pourtahmasbi" BUNDLE_ID org.remsound.plugin
    PLUGIN_MANUFACTURER_CODE Ednp PLUGIN_CODE Rsmc
    FORMATS AU VST3 PRODUCT_NAME "RemSound"
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Fx Network)
juce_add_binary_data(RemSoundPluginManual NAMESPACE RemSoundManual SOURCES ../README.md ../Manual.txt)
juce_add_binary_data(RemSoundPluginHelpSounds NAMESPACE RemSoundHelpSounds HEADER_NAME PluginHelpSoundData.h SOURCES
    "../App/Resources/default sounds/help open 1.wav" "../App/Resources/default sounds/help open 2.wav"
    "../App/Resources/default sounds/help open 3.wav" "../App/Resources/default sounds/help close 1.wav"
    "../App/Resources/default sounds/help close 2.wav" "../App/Resources/default sounds/help close 3.wav")
set_source_files_properties(Source/PluginHelp.mm Source/PluginHelpSounds.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
target_sources(RemSoundPlugin PRIVATE Source/Processor.cpp Source/BridgeClient.cpp Source/PluginHelp.mm Source/PluginHelpSounds.mm)
target_link_libraries(RemSoundPlugin PRIVATE RemSoundPluginManual RemSoundPluginHelpSounds RemSoundPluginHostRuntime "-framework AVFoundation")
target_compile_definitions(RemSoundPlugin PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(RemSoundPlugin PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
clap_juce_extensions_plugin(TARGET RemSoundPlugin CLAP_ID "org.remsound.plugin" CLAP_FEATURES audio-effect utility stereo)
foreach(format RemSoundPlugin_AU RemSoundPlugin_VST3 RemSoundPlugin_CLAP)
    set_target_properties(${format} PROPERTIES INSTALL_RPATH "@loader_path/../Frameworks")
endforeach()
